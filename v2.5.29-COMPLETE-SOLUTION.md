**JavaScript to Add (in product-meta-box-v2.js):**
```javascript
// Handle metal selection change - show/hide making charges and wastage based on metal group settings
jQuery('#jpc_metal_id').on('change', function() {
    var selectedOption = jQuery(this).find('option:selected');
    var enableMaking = selectedOption.data('enable-making');
    var enableWastage = selectedOption.data('enable-wastage');
    
    // Show/hide making charges section
    if (enableMaking == 1) {
        jQuery('.jpc-making-charges-section').show();
    } else {
        jQuery('.jpc-making-charges-section').hide();
    }
    
    // Show/hide wastage field
    if (enableWastage == 1) {
        jQuery('.jpc-wastage-field').show();
    } else {
        jQuery('.jpc-wastage-field').hide();
    }
});

// Trigger on page load
jQuery(document).ready(function() {
    jQuery('#jpc_metal_id').trigger('change');
});
```

#### 3. Price Calculator - Skip Disabled Charges
**File:** `includes/class-jpc-price-calculator-v2.php`
**Method:** `calculate_price()`

**Find the making charge calculation section and wrap it:**
```php
// Get metal group settings
$metal_group = null;
if ($metal && $metal->metal_group_id) {
    $metal_group = JPC_Metal_Groups::get_by_id($metal->metal_group_id);
}

// Calculate making charges only if enabled in metal group
$making_charge = 0;
if (!$metal_group || $metal_group->enable_making_charge) {
    if ($making_charges_mode === 'auto') {
        // Auto calculation
        $making_charge = $metal_weight * ($metal->making_charges_per_gram ?? 0);
    } else {
        // Manual calculation
        if ($making_charges_type === 'percentage') {
            $making_charge = ($metal_cost * $making_charges_value) / 100;
        } else {
            $making_charge = $making_charges_value;
        }
    }
}

// Calculate wastage only if enabled in metal group
$wastage_charge = 0;
if (!$metal_group || $metal_group->enable_wastage_charge) {
    $wastage_charge = ($metal_cost * $wastage) / 100;
}
```
